# ============================================================================
#  MailArchiver — обязательные зависимости
#
#  Устанавливаются скриптом install.sh/update.sh в изолированное окружение
#  (venv) ИМЕННО ИЗ ЭТОГО ФАЙЛА, после чего сам пакет ставится с --no-deps.
#  Так набор библиотек не «уплывает» при каждой переустановке.
#
#  ПОЧЕМУ ДИАПАЗОНЫ, А НЕ ЖЁСТКИЕ ВЕРСИИ.
#  Жёсткие пины (==) ломали установку на новых Python: колёс под свежий
#  интерпретатор для старой версии ещё нет, и pip уходит в сборку из исходников.
#  Так, PyYAML 6.0.2 не ставится на Python 3.14 — колёса cp314 появились только
#  в 6.0.3. Поэтому нижняя граница = минимальная проверенная версия, верхняя =
#  следующий мажорный выпуск, который может сломать совместимость.
#
#  Проверено на Python 3.10–3.14.
# ============================================================================

# Веб-фреймворк и сервер.
# Верхняя граница <1.0 — страховка: в Starlette 0.29 уже менялась сигнатура
# TemplateResponse (request передаётся первым аргументом). Код это учитывает
# (см. mailarchiver/web/app.py: render_template), но мажорный выпуск стоит
# сначала проверить тестами, а потом поднимать границу.
fastapi>=0.115.6,<1.0
uvicorn[standard]>=0.34.0,<1.0
python-multipart>=0.0.20,<1.0

# Шаблоны. 3.1.6 закрывает уязвимость выхода из песочницы (CVE-2025-27516).
jinja2>=3.1.6,<4

# Планировщик. Граница <4 НЕСУЩАЯ: в APScheduler 4.x нет ни BackgroundScheduler
# в текущем виде, ни атрибута job.next_run_time, на которых построен
# mailarchiver/scheduler/scheduler.py. Снимать только вместе с его переписыванием.
apscheduler>=3.11.3,<4

# IMAP. Граница <4: мажорная версия вышла, но на ней сервис не тестировался,
# а через этот клиент проходят все операции с почтой.
IMAPClient>=3.0.1,<4

# Шифрование секретов (Fernet). Верхней границы намеренно нет: пакет публикует
# abi3-колёса, которые ставятся на любой более новый CPython, а используемый
# API (Fernet) стабилен. Ограничение здесь мешало бы установке на новый Python.
cryptography>=44.0

# YAML-конфигурация. 6.0.3 — первая версия с колёсами под Python 3.14.
PyYAML>=6.0.3,<7
